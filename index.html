3-Domain Bloom's Taxonomy Action Verbs Matrix

1. Cognitive (Thinking)

C1: Remember - Define, List, State

C2: Understand - Explain, Discuss, Describe

C3: Apply - Demonstrate, Calculate, Solve

C4: Analyze - Compare, Differentiate, Diagram

C5: Evaluate - Critique, Justify, Assess

C6: Create - Design, Formulate, Construct

2. Affective (Feeling/Values)

A1: Receiving - Attend, Acknowledge, Listen

A2: Responding - Discuss, Conform, Assist

A3: Valuing - Demonstrate, Commit, Promote

A4: Organizing - Formulate, Integrate, Synthesize

A5: Characterizing - Internalize, Exemplify, Act

3. Psychomotor (Physical/Skills)

P1: Perception - Detect, Differentiate, Isolate

P2: Set - Position, Display, Proceed

P3: Guided Response - Copy, Trace, Reproduce

P4: Mechanism - Assemble, Calibrate, Operate

P5: Complex Response - Build, Master, Execute

P6: Adaptation - Adapt, Reorganize, Vary

P7: Origination - Construct, Create, Produce

-- PostgreSQL Schema with Bloom 3 Domains & Certificate Support
CREATE TABLE IF NOT EXISTS public.submissions (
    id TEXT PRIMARY KEY,
    user_id UUID REFERENCES auth.users(id),
    participant_name TEXT NOT NULL,
    institution TEXT NOT NULL,
    course_code TEXT NOT NULL,
    course_title TEXT NOT NULL,
    total_slt NUMERIC NOT NULL,
    score NUMERIC NOT NULL,
    status TEXT DEFAULT 'Submitted',
    feedback TEXT,
    verification_id TEXT,
    data JSONB NOT NULL,
    submitted_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);
ALTER TABLE public.submissions ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Public submission view" ON public.submissions FOR SELECT USING (true);
CREATE POLICY "Participant submission insert" ON public.submissions FOR INSERT WITH CHECK (true);
